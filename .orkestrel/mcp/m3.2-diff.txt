+	})
+	await client.connect()
+

+	if (id === undefined) throw new Error('Expected the guide call to carry an id')

+		method: 'notifications/progress',

+			_meta: { [MCP_META_SUBSCRIPTION]: 9_999 },
+		},
+	})
+	transport.receive({
