+	// operation's one-source derivation chain. Each candidate's own span is read BEFORE
+	// its derivation entry is followed, and a visited set stops the walk on a cycle. That
+	// ordering is load-bearing: a handler that hands back an original node gives that
+	// object an output entry pointing at the slot it was placed in, so reading the span
+	// first terminates the chain on the node's own region instead of following that entry
+	// onto a foreign one - both when the node is being resolved and when a later hop
+	// reaches it as someone else's source.
+	#derive(

+		derivations: ReadonlyMap<MarkdownNode, MarkdownNode | undefined>,
+	): Markdown {
